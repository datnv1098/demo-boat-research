1. Sử dụng "Public IP" cho ssh và config DNS
2. EC2 inbound mở port 3000 và 80

// WINDOW
cmd /c "npm run build"
scp -i "key\EC2PEM.pem" -o StrictHostKeyChecking=no -r dist/* ubuntu@18.141.158.6:/tmp/frontend/
scp -i "key\EC2PEM.pem" -o StrictHostKeyChecking=no server.js ubuntu@18.141.158.6:/tmp/
ssh -i "key\EC2PEM.pem" -o StrictHostKeyChecking=no ubuntu@18.141.158.6 "sudo rm -rf /var/www/fisheries-demo/dist && sudo mkdir -p /var/www/fisheries-demo/dist && sudo mv /tmp/frontend/* /var/www/fisheries-demo/dist/ && sudo mv /tmp/server.js /var/www/fisheries-demo/ && pm2 restart fisheries-demo && sleep 3 && curl -I http://localhost:3000 && echo 'Application is running!'"
